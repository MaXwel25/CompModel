    vector<vector<double>> streams(n);
    
    for (int i = 0; i < n; i++)
    {
        long double time = 0.0;
            while (time < T)
            {
                double interval = dist(mt);
                time += interval;
                if (time <= T)
                {
                    streams[i].push_back(time);
                }
            }
    }

        }
